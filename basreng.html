Tindakan Wajib Pasca-Restore/Reset (Reset Sequence)

Jika Anda melakukan Overwrite atau Reset, urutan ID (auto-increment) di database Supabase bisa bertabrakan. Salin script SQL di bawah ini dan jalankan di menu SQL Editor pada dashboard Supabase Anda. 


-- Reset urutan ID untuk tabel Pencatatan Masuk
SELECT setval(pg_get_serial_sequence('pencatatan_basreng', 'id'), COALESCE((SELECT MAX(id) FROM pencatatan_basreng), 0) + 1, false);

-- Reset urutan ID untuk tabel Nota Keluar
SELECT setval(pg_get_serial_sequence('nota_keluar', 'id'), COALESCE((SELECT MAX(id) FROM nota_keluar), 0) + 1, false);

-- Reset urutan ID untuk Riwayat Invoice
SELECT setval(pg_get_serial_sequence('riwayat_invoice', 'id'), COALESCE((SELECT MAX(id) FROM riwayat_invoice), 0) + 1, false);

-- Reset urutan ID untuk Riwayat Fee Kiriman
SELECT setval(pg_get_serial_sequence('riwayat_fee', 'id'), COALESCE((SELECT MAX(id) FROM riwayat_fee), 0) + 1, false);

-- Reset urutan ID untuk Riwayat Fee Plastik
SELECT setval(pg_get_serial_sequence('riwayat_fee_plastik', 'id'), COALESCE((SELECT MAX(id) FROM riwayat_fee_plastik), 0) + 1, false);

-- Reset urutan ID untuk Titip Uang (Deposit)
SELECT setval(pg_get_serial_sequence('titip_uang', 'id'), COALESCE((SELECT MAX(id) FROM titip_uang), 0) + 1, false);
